Up in three steps.

All you need is Docker with the compose plugin.

Try it first
# 1. Get the compose file and the settings$ curl -O https://raw.githubusercontent.com/laupernoe/plimail/main/docker-compose.yml$ curl -o .env https://raw.githubusercontent.com/laupernoe/plimail/main/.env.example# 2. In .env, set APP_URL and these secrets$ openssl rand -hex 32   # SESSION_SECRET, ENCRYPTION_KEY$ openssl rand -hex 16   # DB_PASSWORD# 3. Start$ docker compose up -d

Open http://your-host:3000 and create the first account. It becomes the admin.

docker-compose.yml the whole file, as in the repository 
services:
  plimail:
    image: ghcr.io/laupernoe/plimail:latest
    build: .
    restart: unless-stopped
    # Room for a send in flight to finish on a restart (the app waits up to 25 s).
    stop_grace_period: 30s
    env_file: .env
    ports:
      - "${HTTP_PORT:-3000}:3000"
      - "${HTTPS_PORT:-3443}:3443"
    volumes:
      - image_cache:/var/cache/mailflow/images
      # The self-signed certificate, generated at the first start. To serve your own, bind-mount a
      # directory holding cert.pem and key.pem here instead.
      - certs:/certs
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy

  postgres:
    image: postgres:16-alpine
    restart: unless-stopped
    environment:
      POSTGRES_DB: ${DB_NAME:-mailflow}
      POSTGRES_USER: ${DB_USER:-mailflow}
      POSTGRES_PASSWORD: ${DB_PASSWORD:?set DB_PASSWORD in .env}
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U \"$$POSTGRES_USER\" -d \"$$POSTGRES_DB\""]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 20s

  redis:
    image: redis:7-alpine
    restart: unless-stopped
    command: redis-server --save 60 1 --loglevel warning
    volumes:
      - redis_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 3s
      retries: 5
      start_period: 10s

# The volume names are the ones earlier versions used, so an existing install keeps its data.
volumes:
  postgres_data:
  redis_data:
  image_cache:
  certs:
